\section{Método de bisección}
Sea $f(x) = x^3 - 2x - 5$ continua en $[a,b]$ con $f(a) \cdot f(b) < 0$.
El punto medio es $c = \frac{a+b}{2}$.

% comentario: no debe aparecer
\subsection{Error}
El error después de $n$ iteraciones cumple
\[ |x_n - r| \le \frac{b-a}{2^{n}} \]

\begin{enumerate}
  \item Calcular $c_1 = \frac{a_0 + b_0}{2}$.
  \item Si $f(c_1) = 0$, terminar.
  \item Repetir con $[a_1, b_1]$.
\end{enumerate}

\section{Fórmulas útiles}
\begin{align*}
  S &= \sum_{k=0}^{n} \frac{1}{k!} \\
  I &= \int_0^1 x^2 \, dx = \frac{1}{3} \\
  \log_2(8) &= 3, \quad \sqrt[3]{27} = 3
\end{align*}
Matriz: $A = \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}$ y $|A| = -2$.

Errores: $\alpha \approx 0.5$, $\bar{x}$, $\hat{y}$, $e^{-x}$ y $\epsilon_{mach}$.
\note{Tip}{Usa $\varepsilon = 10^{-6}$.\\ Detente si $|b-a| < \varepsilon$.}
